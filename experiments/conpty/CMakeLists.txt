target_compile_definitions(termforge_conpty_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)

  target_compile_options(termforge_conpty_probe PRIVATE /W4 /EHsc)

  target_compile_options(termforge_conpty_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
