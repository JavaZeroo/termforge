if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.20)
  project(termforge_conpty LANGUAGES CXX)
endif()
if(NOT WIN32)
  return()
endif()
add_executable(termforge_conpty_probe probe.cpp)
target_compile_features(termforge_conpty_probe PRIVATE cxx_std_17)
target_compile_definitions(termforge_conpty_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
if(MSVC)
  target_compile_options(termforge_conpty_probe PRIVATE /W4 /EHsc)
else()
  target_compile_options(termforge_conpty_probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
