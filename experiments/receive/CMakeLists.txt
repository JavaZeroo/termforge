cmake_minimum_required(VERSION 3.20)
project(termforge_receive LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This experiment uses Windows anonymous pipes")
endif()
add_executable(receive receive.cpp)
target_compile_features(receive PRIVATE cxx_std_17)
target_compile_definitions(receive PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(receive PRIVATE /W4 /O2)
target_link_libraries(receive PRIVATE Psapi)
add_executable(replay ../core/replay.cpp)
target_compile_features(replay PRIVATE cxx_std_17)
target_compile_options(replay PRIVATE /W4 /O2)
