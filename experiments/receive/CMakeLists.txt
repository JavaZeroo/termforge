cmake_minimum_required(VERSION 3.20)

# Retain the old standalone experiment command; root CMake is the supported CI entry.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(termforge_receive LANGUAGES CXX)
  function(termforge_warnings target)
    target_compile_features(${target} PRIVATE cxx_std_17)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endfunction()
  add_executable(replay ../core/replay.cpp)
  termforge_warnings(replay)
endif()

if(WIN32)
  add_executable(receive receive.cpp)
  termforge_warnings(receive)
  target_compile_definitions(receive PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(receive PRIVATE Psapi)
endif()
